b. Count repeated characters in a string

Code:

# Count repeated characters in a string
input_string = "programming is fun"
char_count = {}

# Count occurrences of each character
for char in input_string:
    if char in char_count:
        char_count[char] += 1
    else:
        char_count[char] = 1

# Filter only repeated characters
repeated_chars = {char: count for char, count in char_count.items() if count > 1}

print(f"String: '{input_string}'")
print(f"Repeated characters: {repeated_chars}")

Output:

String: 'programming is fun'
Repeated characters: {'r': 2, 'g': 2, 'm': 2, 'i': 2, 'n': 2}

def reverse_string(input_string):
    return input_string[::-1]

# Example usage

d. List operations (add, remove, replace, remove duplicates)

Code:

def list_operations():
    # Create a list
    my_list = ['apple', 'banana', 'cherry', 'date', 'elderberry', 'fig', 'grape']
    print(f"Original list: {my_list}")
    
    # Add an item in the 4th position (index 3)
    my_list.insert(3, 'dragonfruit')
    print(f"After adding 'dragonfruit' at 4th position: {my_list}")
    
    # Remove the 3rd item (index 2)
    removed_item = my_list.pop(2)
    print(f"After removing 3rd item ('{removed_item}'): {my_list}")
    
    # Replace the 6th item (index 5) with new item
    my_list[5] = 'kiwi'
    print(f"After replacing 6th item with 'kiwi': {my_list}")
    
    # Add duplicates to show removal
    my_list.extend(['apple', 'kiwi', 'banana'])
    print(f"List with duplicates: {my_list}")
    
    # Remove duplicates
    my_list = list(dict.fromkeys(my_list))
    print(f"After removing duplicates: {my_list}")
    
    return my_list

Output:

Original list: ['apple', 'banana', 'cherry', 'date', 'elderberry', 'fig', 'grape']
After adding 'dragonfruit' at 4th position: ['apple', 'banana', 'cherry', 'dragonfruit', 'date', 'elderberry', 'fig', 'grape']
After removing 3rd item ('cherry'): ['apple', 'banana', 'dragonfruit', 'date', 'elderberry', 'fig', 'grape']
After replacing 6th item with 'kiwi': ['apple', 'banana', 'dragonfruit', 'date', 'elderberry', 'kiwi', 'grape']
List with duplicates: ['apple', 'banana', 'dragonfruit', 'date', 'elderberry', 'kiwi', 'grape', 'apple', 'kiwi', 'banana']
After removing duplicates: ['apple', 'banana', 'dragonfruit', 'date', 'elderberry', 'kiwi', 'grape']

f. Find common elements between two lists

Code:

def find_common_elements():
    # Create two lists
    list1 = [1, 2, 3, 4, 5, 6]
    list2 = [4, 5, 6, 7, 8, 9]
    
    print(f"First list: {list1}")
    print(f"Second list: {list2}")
    
    # Find common elements
    common_elements = [item for item in list1 if item in list2]
    
    print(f"Common elements: {common_elements}")
    
    return common_elements

Output:

First list: [1, 2, 3, 4, 5, 6]
Second list: [4, 5, 6, 7, 8, 9]
Common elements: [4, 5, 6]

g. Remove repeated items from a list

Code:

def remove_repeated_items():
    # Create a list with repeated items
    original_list = [2, 3, 4, 3, 0, 0, 5, 5, 9, 9]
    print(f"Original list: {original_list}")
    
    # Remove duplicates while preserving order
    result_list = []
    for item in original_list:
        if item not in result_list:
            result_list.append(item)
    
    print(f"List after removing duplicates: {result_list}")
    
    return result_list

Output:

Original list: [2, 3, 4, 3, 0, 0, 5, 5, 9, 9]
List after removing duplicates: [2, 3, 4, 0, 5, 9]

h. Copy list without reflecting changes

Code:

import copy

def copy_list():
    # Create original list
    list1 = [1, 2, [3, 4], 5]
    print(f"Original list1: {list1}")
    
    # Copy using copy() method (shallow copy)
    list2 = list1.copy()
    
    # For deep copy (to handle nested lists)
    list2_deep = copy.deepcopy(list1)
    
    # Modify list1
    list1.append(6)       # Add an element
    list1[2].append(3.5)  # Modify nested list
    
    print(f"Modified list1: {list1}")
    print(f"list2 (shallow copy): {list2}")
    print(f"list2_deep (deep copy): {list2_deep}")
    
    return list2_deep

Output:

Original list1: [1, 2, [3, 4], 5]
Modified list1: [1, 2, [3, 4, 3.5], 5, 6]
list2 (shallow copy): [1, 2, [3, 4, 3.5], 5]
list2_deep (deep copy): [1, 2, [3, 4], 5]

Dictionary and Set Operations

a. Sort a dictionary by value

Code:

def sort_dict_by_value():
    # Create a dictionary
    my_dict = {'apple': 25, 'banana': 10, 'cherry': 40, 'date': 5, 'elderberry': 30}
    print(f"Original dictionary: {my_dict}")
    
    # Sort dictionary by value
    sorted_dict = dict(sorted(my_dict.items(), key=lambda item: item[1]))
    print(f"Dictionary sorted by value (ascending): {sorted_dict}")
    
    # Sort dictionary by value in descending order
    sorted_dict_desc = dict(sorted(my_dict.items(), key=lambda item: item[1], reverse=True))
    print(f"Dictionary sorted by value (descending): {sorted_dict_desc}")
    
    return sorted_dict

Output:

Original dictionary: {'apple': 25, 'banana': 10, 'cherry': 40, 'date': 5, 'elderberry': 30}
Dictionary sorted by value (ascending): {'date': 5, 'banana': 10, 'apple': 25, 'elderberry': 30, 'cherry': 40}
Dictionary sorted by value (descending): {'cherry': 40, 'elderberry': 30, 'apple': 25, 'banana': 10, 'date': 5}

d. Create a union of sets

Code:

def union_of_sets():
    # Create sets
    set1 = {1, 2, 3, 4, 5}
    set2 = {4, 5, 6, 7, 8}
    
    print(f"Set1: {set1}")
    print(f"Set2: {set2}")
    
    # Method 1: Using the union() method
    union_result = set1.union(set2)
    print(f"Union using union() method: {union_result}")
    
    # Method 2: Using the | operator
    union_result_alt = set1 | set2
    print(f"Union using | operator: {union_result_alt}")
    
    return union_result

Output:

Set1: {1, 2, 3, 4, 5}
Set2: {4, 5, 6, 7, 8}
Union using union() method: {1, 2, 3, 4, 5, 6, 7, 8}
Union using | operator: {1, 2, 3, 4, 5, 6, 7, 8}

e. Match a string with 'a' followed by one or more b's

Code:

import re

def match_pattern():
    # Pattern: 'a' followed by one or more 'b's
    pattern = r'ab+'
    
    # Test strings
    test_strings = [
        "ab",
        "abb",
        "abbb",
        "a",
        "b",
        "abc",
        "cab",
        "Hello abbb world"
    ]
    
    print("Testing pattern: 'a' followed by one or more b's")
    print("Pattern: ab+")
    
    for test_str in test_strings:
        match = re.search(pattern, test_str)
        if match:
            print(f"'{test_str}' MATCHES - Found '{match.group()}'")
        else:
            print(f"'{test_str}' DOES NOT MATCH")

# Run the function
match_pattern()

Output:

Testing pattern: 'a' followed by one or more b's
Pattern: ab+
'ab' MATCHES - Found 'ab'
'abb' MATCHES - Found 'abb'
'abbb' MATCHES - Found 'abbb'
'a' DOES NOT MATCH
'b' DOES NOT MATCH
'abc' MATCHES - Found 'ab'
'cab' MATCHES - Found 'ab'
'Hello abbb world' MATCHES - Found 'abbb'
